local UsedDeviceFile = "BrakyoHub_DeviceLocked.json"
local ActiveKeyFile = "BrakyoHub_ActiveSession.json"

local function IsDeviceLocked()
    if not (readfile and isfile) then return false end
    return isfile(UsedDeviceFile)
end

local function LockDevicePermanently()
    if writefile then
        pcall(function()
            writefile(UsedDeviceFile, game:GetService("HttpService"):JSONEncode({LockedTime = os.time()}))
        end)
    end
end

local function CheckActiveSession()
    if not (readfile and writefile and isfile and delfile) then return false end
    if isfile(ActiveKeyFile) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(ActiveKeyFile))
        end)
        if success and data and data.ExpireTime then
            if os.time() < data.ExpireTime then
                return true
            else
                delfile(ActiveKeyFile)
            end
        end
    end
    return false
end

SubmitBtn.MouseButton1Click:Connect(function()
    local enteredKey = KeyBox.Text
    
    if enteredKey == SpecialKey then
        if IsDeviceLocked() then
            StatusLabel.TextColor3 = Color3.fromRGB(255, 90, 90)
            StatusLabel.Text = "❌ Thiết bị này đã dùng key đặc biệt này rồi và không thể dùng lại nữa!"
            return
        end
        
        LockDevicePermanently()
        if writefile then
            local expireTime = os.time() + SpecialKeyDuration
            writefile(ActiveKeyFile, game:GetService("HttpService"):JSONEncode({ExpireTime = expireTime}))
        end
        
        StatusLabel.TextColor3 = Color3.fromRGB(0, 220, 220)
        StatusLabel.Text = "✅ Thành công! Kích hoạt 6 tiếng."
        task.wait(0.8)
        KeySystemFrame:Destroy()
        DisclaimerFrame.Visible = true
        
    elseif ValidKeys[enteredKey] then
        if writefile then
            local expireTime = os.time() + NormalKeyDuration
            writefile(ActiveKeyFile, game:GetService("HttpService"):JSONEncode({ExpireTime = expireTime}))
        end
        
        StatusLabel.TextColor3 = Color3.fromRGB(0, 220, 220)
        StatusLabel.Text = "✅ Thành công! Kích hoạt 12 tiếng."
        task.wait(0.8)
        KeySystemFrame:Destroy()
        DisclaimerFrame.Visible = true
    else
        StatusLabel.TextColor3 = Color3.fromRGB(255, 90, 90)
        StatusLabel.Text = "❌ Key không chính xác!"
    end
end)
