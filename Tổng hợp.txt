-- Hàm kiểm tra xem key đã lưu còn hạn hay không
local function CheckSavedKey()
    if not (readfile and writefile and isfile and delfile) then return false end
    if isfile(FileName) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(FileName))
        end)
        if success and data and data.ExpireTime then
            if os.time() < data.ExpireTime then
                return true -- Key vẫn còn hạn
            else
                delfile(FileName) -- Hết hạn thì xóa file lưu cũ
            end
        end
    end
    return false
end

        MinButton.Visible = true

-- Hàm phụ trợ để thêm nút script con bên trong trang
local function AddScriptButton(targetPage, scriptName, callback)
    local btn = Instance.new("TextButton")
    btn.Parent = targetPage
    btn.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
    btn.Size = UDim2.new(1, -10, 0, 36)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = "  " .. scriptName
    btn.TextColor3 = Color3.fromRGB(230, 230, 230)
    btn.TextSize = 13
    btn.TextXAlignment = Enum.TextXAlignment.Left
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn
    
    btn.MouseButton1Click:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(0, 255, 255), TextColor3 = Color3.fromRGB(15, 15, 20)}):Play()
        task.wait(0.1)
        TweenService:Create(btn, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(30, 30, 40), TextColor3 = Color3.fromRGB(230, 230, 230)}):Play()
        pcall(callback)
    end)
    
    targetPage.CanvasSize = UDim2.new(0, 0, 0, targetPage.UIListLayout.AbsoluteContentSize.Y + 10)
end

-- Tính năng tìm kiếm động
SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
    local text = SearchBox.Text:lower()
    for _, page in pairs(Pages) do
        for _, child in pairs(page:GetChildren()) do
            if child:IsA("TextButton") then
                if text == "" or child.Text:lower():find(text) then
                    child.Visible = true
                else
                    child.Visible = false
                end
            end
        end
    end
end)

-- Danh sách lưu trữ các nút menu trái
local MenuButtons = {}

local function CreateMenuButton(name, targetPage)
    local btn = Instance.new("TextButton")
    btn.Parent = LeftMenu
    btn.BackgroundColor3 = Color3.fromRGB(22, 22, 30)
    btn.Size = UDim2.new(1, -5, 0, 36)
    btn.Font = Enum.Font.GothamMedium
    btn.Text = "  " .. name
    btn.TextColor3 = Color3.fromRGB(170, 170, 190)
    btn.TextSize = 13
    btn.TextXAlignment = Enum.TextXAlignment.Left
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn
    
    table.insert(MenuButtons, {Button = btn, Page = targetPage})
    
    btn.MouseButton1Click:Connect(function()
        for _, p in pairs(Pages) do
            p.Visible = false
        end
        targetPage.Visible = true
        
        for _, item in ipairs(MenuButtons) do
            TweenService:Create(item.Button, TweenInfo.new(0.2), {
                BackgroundColor3 = Color3.fromRGB(22, 22, 30),
                TextColor3 = Color3.fromRGB(170, 170, 190),
                Size = UDim2.new(1, -5, 0, 36)
            }):Play()
        end
        
        TweenService:Create(btn, TweenInfo.new(0.15), {
            BackgroundColor3 = Color3.fromRGB(35, 35, 50),
            TextColor3 = Color3.fromRGB(0, 255, 255),
            Size = UDim2.new(1, -9, 0, 34)
        }):Play()
        
        task.wait(0.15)
        TweenService:Create(btn, TweenInfo.new(0.15), {
            Size = UDim2.new(1, -5, 0, 36)
        }):Play()
    end)
end
